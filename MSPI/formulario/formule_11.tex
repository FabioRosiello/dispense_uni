% Formulario, argomento 11: variabili continue notevoli (dispensa, capitolo 11; varianze dal capitolo 13)
% ============================================================
\argomento{Le variabili continue notevoli}{Dispensa: capitolo 11 (le varianze sono nel capitolo 13)}[Variabili
  continue notevoli]

\voce{uniformec}{Variabile uniforme continua $\mathcal{U}(a, b)$}{nessun punto privilegiato in $(a, b)$}
  {f_X(x) = \frac{1}{b - a} \ \text{ per } x \in (a, b), \quad 0 \ \text{altrove}}
  {\variante{CDF}{F_X(x) = \frac{x - a}{b - a} \ \text{ per } a \leq x \leq b \quad (0 \text{ prima, } 1 \text{ dopo})}
   \variante{media e varianza}{E[X] = \frac{a + b}{2}, \quad \sigma_X^2 = \frac{(b - a)^2}{12}, \quad
                               E[X^2] = \frac{a^2 + ab + b^2}{3}}
   \variante{con il rettangolo}{\mathcal{U}(0, 1): \ f_X(x) = \mathrm{rect}\left(x - \tfrac{1}{2}\right)}}

\voce{esponenziale}{Variabile esponenziale $\mathcal{E}(\lambda)$}{tempi di attesa, $\lambda > 0$}
  {f_X(x) = \lambda\, e^{-\lambda x}\, u(x)}
  {\variante{CDF e CCDF}{F_X(x) = \left(1 - e^{-\lambda x}\right) u(x), \quad P(X > x) = e^{-\lambda x} \ (x \geq 0)}
   \variante{media e varianza}{E[X] = \frac{1}{\lambda}, \quad \sigma_X^2 = \frac{1}{\lambda^2}, \quad
                               E[X^2] = \frac{2}{\lambda^2} \quad (\sigma_X = \mu_X)}}

\voce{memoria}{Assenza di memoria dell'esponenziale}{$X \sim \mathcal{E}(\lambda)$, \ $s, t \geq 0$}
  {P(X > s + t \mid X > s) = P(X > t) = e^{-\lambda t}}
  {}

\voce{laplacevar}{Variabile di Laplace $\mathcal{L}(\lambda)$}{due esponenziali simmetrici, $\lambda > 0$}
  {f_X(x) = \frac{\lambda}{2}\, e^{-\lambda \abs{x}}}
  {\variante{CDF}{F_X(x) = \tfrac{1}{2}\, e^{\lambda x} \ (x \leq 0), \quad F_X(x) = 1 - \tfrac{1}{2}\, e^{-\lambda x} \ (x \geq 0)}
   \variante{media e varianza}{E[X] = 0, \quad \sigma_X^2 = E[X^2] = \frac{2}{\lambda^2}, \quad
                               \sigma_X = \frac{\sqrt{2}}{\lambda}}}

\voce{cauchyvar}{Variabile di Cauchy $\mathcal{C}(a, b)$}{campana con code pesanti, $b > 0$}
  {f_X(x) = \frac{1}{b\pi}\; \frac{1}{1 + \left( \frac{x - a}{b} \right)^2}}
  {\variante{CDF}{F_X(x) = \frac{1}{2} + \frac{1}{\pi} \arctan\left( \frac{x - a}{b} \right)}
   \variante{media e varianza}{\text{non esistono} \quad (E[X^2] = +\infty)}
   \variante{punto di simmetria (valor principale)}
            {\frac{1}{b\pi} \lim_{H \to \infty} \int_{-H}^{H} \frac{x}{1 + \left( \frac{x - a}{b} \right)^2} \dd x = a}}
