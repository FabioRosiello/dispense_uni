% ============================================================
% Formulario, argomento 17: processi aleatori (dispensa, capitolo 17)
% ============================================================
\argomento{Processi aleatori}{Dispensa: capitolo 17}

\voce{catenavettori}{Regola della catena per i vettori}{una variabile alla volta}
  {p_{\vett{X}}(\vett{x}) = \prod_{i=1}^{m} p_{X_i \mid X_1, \ldots, X_{i-1}}(x_i \mid x_1, \ldots, x_{i-1})}
  {\variante{con le densità}{f_{\vett{X}}(\vett{x}) = \prod_{i=1}^{m} f_{X_i \mid X_1, \ldots, X_{i-1}}(x_i \mid x_1, \ldots, x_{i-1})}
   \variante{variabili indipendenti}{p_{\vett{X}}(\vett{x}) = \prod_{i=1}^{m} p_{X_i}(x_i)}}

\voce{processo}{Processo aleatorio a tempo discreto}{una successione per ogni risultato}
  {X : \omega \in \Omega \;\longmapsto\; \{X(n, \omega)\}_{n \in \Z}}
  {\variante{fissato $\omega$}{X(n, \omega) \text{ è una realizzazione (una successione di numeri)}}
   \variante{fissato $n$}{X(n) \text{ è una variabile aleatoria, con densità } f_{X(n)}(x; n)}}

\voce{stazprimo}{Stazionarietà del primo ordine}{la densità di un campione non dipende dall'istante}
  {f_{X(n)}(x; n) = f_X(x) \qquad \text{per ogni } n}
  {}

\voce{stazsecondo}{Stazionarietà del secondo ordine}{conta solo la distanza fra i due istanti}
  {f_{X(n_1), X(n_2)}(x_1, x_2; n_1, n_2) = f_{X(n_1 + h), X(n_2 + h)}(x_1, x_2; n_1 + h, n_2 + h) \qquad \forall h}
  {\variante{implica quella del primo ordine}{}}

\voce{stazstretta}{Stazionarietà in senso stretto}{densità di $M$ campioni invarianti per traslazione}
  {f_{\vett{X}}(x_1, \ldots, x_M; n_1, \ldots, n_M) = f_{\vett{X}}(x_1, \ldots, x_M; n_1 + h, \ldots, n_M + h)
   \qquad \forall h}
  {\variante{in senso stretto}{\text{per ogni ordine } M}
   \variante{l'ordine $M$ implica tutti gli ordini $i \leq M$}{}}

\voce{procindip}{Processo indipendente}{basta il primo ordine}
  {f_{\vett{X}}(x_1, \ldots, x_M; n_1, \ldots, n_M) = \prod_{i=1}^{M} f_{X(n_i)}(x_i; n_i)}
  {\variante{discreto, indipendente e stazionario}{P\big(X(n_1) = x_1, \ldots, X(n_M) = x_M\big) = \prod_{i=1}^{M} p_X(x_i)}}

\voce{autocorrelazione}{Autocorrelazione}{correlazione fra due istanti}
  {R_X(n_1, n_2) = E[X(n_1)\, X(n_2)]}
  {}

\voce{ssl}{Processo stazionario in senso lato (SSL)}{\emph{wide-sense stationary}, WSS}
  {E[X(n)] = \mu_X \qquad R_X(n_1, n_2) = R_X(n_2 - n_1)}
  {\variante{con la distanza $k$}{R_X(k) = E[X(n)\, X(n + k)], \quad R_X(-k) = R_X(k)}
   \variante{potenza media}{R_X(0) = E[X^2(n)]}
   \variante{stazionario del secondo ordine $\Rightarrow$ SSL, non il viceversa}{}}

\voce{toeplitz}{Campioni di un processo SSL}{$M$ campioni, $X_i = X(n_i)$}
  {\vett{\mu}_{\vett{X}} = \mu\, \vett{1} \qquad \matr{C}_{\vett{X}} = \sigma_X^2 \begin{bmatrix} 1 & \rho_{1,2} &
   \cdots & \rho_{1,M} \\ \rho_{2,1} & 1 & \cdots & \rho_{2,M} \\ \vdots & & \ddots & \vdots \\ \rho_{M,1} & \cdots &
   \cdots & 1 \end{bmatrix}, \quad \rho_{i,j} = \rho_{\abs{n_i - n_j}}}
  {\variante{campioni equispaziati}{\matr{C}_{\vett{X}} \text{ è di Toeplitz: ogni diagonale ha elementi uguali}}}

\voce{bernoulliproc}{Processo di Bernoulli $\pm 1$}{indipendente, nessuna memoria}
  {P(X(n) = \pm 1) = \frac{1}{2} \qquad \mu_X = 0 \qquad R_X(k) = \begin{cases} 1 & k = 0 \\ 0 & k \neq 0 \end{cases}}
  {\variante{campioni}{\matr{C}_{\vett{X}} = \matr{I}}
   \variante{quaternario indipendente}{X(n) \in \{-2, -1, 1, 2\}, \quad P(X(n) = i) = \frac{1}{4}}}

\voce{mediamobile}{Media mobile}{$X(n)$ Bernoulli $\pm 1$; memoria di un passo}
  {Y(n) = \frac{1}{2}\left[ X(n) + X(n-1) \right] \qquad
   R_Y(k) = \begin{cases} \frac{1}{2} & k = 0 \\ \frac{1}{4} & k = \pm 1 \\ 0 & \abs{k} \geq 2 \end{cases}}
  {\variante{coefficienti}{\mu_Y = 0, \quad \sigma_Y^2 = \frac{1}{2}, \quad \rho_1 = \frac{1}{2}, \quad \rho_k = 0 \ (\abs{k} \geq 2)}}

\voce{sinusoideproc}{Sinusoide con fase a caso}{$\Phi$ uniforme su un periodo}
  {X(n) = A\cos(2\pi f_0 n + \Phi) \qquad \mu_X = 0 \qquad R_X(k) = \frac{A^2}{2} \cos(2\pi f_0 k)}
  {\variante{densità del primo ordine (arcoseno, stazionaria)}{f_{X(n)}(x) = \frac{1}{\pi\sqrt{A^2 - x^2}}, \quad \abs{x} < A}
   \variante{potenza media}{R_X(0) = \frac{A^2}{2}}}

\voce{passeggiata}{Passeggiata aleatoria}{$X(i)$ Bernoulli $\pm 1$; non stazionaria}
  {S(n) = X(1) + \cdots + X(n) \qquad E[S(n)] = 0 \qquad \sigma_{S(n)}^2 = n}
  {\variante{autocorrelazione, che non dipende solo da $n_2 - n_1$}{R_S(n_1, n_2) = \min(n_1, n_2)}}
