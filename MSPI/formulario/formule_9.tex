% Formulario, argomento 9: funzioni di più variabili, covarianza e correlazione (dispensa, capitolo 9)

\argomento{Funzioni di più variabili, covarianza e correlazione}{Dispensa: capitolo 9}[Covarianza e correlazione]

\voce{pmffunzione2}{Pmf di una funzione $Z = g(X, Y)$}{}
  {p_Z(z) = \sum_{(x, y) \,:\, g(x, y) = z} p_{X,Y}(x, y)}
  {\variante{$g$ biunivoca sulle coppie}{p_Z(z) = p_{X,Y}\left[x(z), y(z)\right]}}

\voce{valoremedio2}{Teorema fondamentale per due variabili}{}
  {E[g(X, Y)] = \sum_{x \in \mathcal{X}} \sum_{y \in \mathcal{Y}} g(x, y)\, p_{X,Y}(x, y)}
  {}

\voce{linearita}{Linearità della media}{vale sempre, anche per variabili dipendenti}
  {E[aX + bY] = a\,E[X] + b\,E[Y]}
  {\variante{$m$ variabili}{E\left[\sum_{i=1}^{m} a_i X_i\right] = \sum_{i=1}^{m} a_i\, E[X_i]}}

\voce{mediacondizionata}{Teorema della media condizionata}{la media in due tempi}
  {E[g(X, Y)] = E\big[\, E[g(X, Y) \mid Y] \,\big] = \sum_{y \in \mathcal{Y}} p_Y(y)\, h(y)}
  {\variante{con}{h(y) = E[g(X, Y) \mid Y = y] = \sum_{x \in \mathcal{X}} g(x, y)\, p_{X \mid Y}(x \mid y)}}

\voce{correlazione}{Correlazione}{media del prodotto}
  {R_{XY} = E[XY] = \sum_{x \in \mathcal{X}} \sum_{y \in \mathcal{Y}} x\, y\, p_{X,Y}(x, y)}
  {}

\voce{covarianza}{Covarianza}{quanto $X$ e $Y$ si muovono insieme}
  {COV(X, Y) = \sigma_{XY}^2 = E\left[(X - \mu_X)(Y - \mu_Y)\right] = E[XY] - E[X]\, E[Y]}
  {\variante{se una delle due ha media nulla}{COV(X, Y) = R_{XY}}
   \variante{si spezza sulle somme}{COV(U + V, W) = COV(U, W) + COV(V, W)}}

\voce{incorrelate}{Variabili incorrelate}{indipendenti $\Rightarrow$ incorrelate, non il viceversa}
  {COV(X, Y) = 0 \iff E[XY] = E[X]\, E[Y]}
  {}

\voce{cauchy}{Limite della covarianza}{$\sigma_X, \sigma_Y > 0$}
  {\abs{COV(X, Y)} \leq \sigma_X\, \sigma_Y}
  {}

\voce{rho}{Coefficiente di correlazione}{misura la dipendenza lineare}
  {\rho_{XY} = \frac{COV(X, Y)}{\sigma_X\, \sigma_Y} \in [-1, 1]}
  {\variante{$\rho_{XY} = \pm 1$ esattamente quando}{Y = aX + b}}

\voce{varsomma}{Varianza di una combinazione lineare}{$Z = aX + bY$}
  {\sigma_Z^2 = a^2 \sigma_X^2 + b^2 \sigma_Y^2 + 2ab\, COV(X, Y) \qquad \mu_Z = a\mu_X + b\mu_Y}
  {\variante{$X$ e $Y$ incorrelate (per esempio indipendenti)}{\sigma_{X+Y}^2 = \sigma_X^2 + \sigma_Y^2}
   \variante{differenza}{\sigma_{X-Y}^2 = \sigma_X^2 + \sigma_Y^2 - 2\, COV(X, Y)}
   \variante{$N$ variabili indipendenti}{\sigma_{X_1 + \cdots + X_N}^2 = \sigma_{X_1}^2 + \cdots + \sigma_{X_N}^2}}
