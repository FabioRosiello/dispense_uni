\argomento{Informazione ed entropia}{Dispensa: capitolo 25}

\voce{informazione}{Informazione di un evento}{più l'evento è raro, più informa}
  {i(A) = \log \frac{1}{P(A)}}
  {\variante{in bit}{i(A) = \log_2 \frac{1}{P(A)}}
   \variante{eventi indipendenti}{i(A \cap B) = i(A) + i(B)}
   \variante{non negativa e decrescente in $P(A)$; l'evento certo non informa}{i(\Omega) = \log 1 = 0}}

\voce{entropia}{Entropia di una variabile discreta}{l'informazione media: l'incertezza a priori}
  {H(X) = E[i(X)] = \sum_{i=1}^{M} p_X(x_i) \log \frac{1}{p_X(x_i)}}
  {\variante{unità di misura}{\text{bit con il logaritmo in base 2, nat con quello in base } e}
   \variante{un valore con probabilità nulla conta zero}{p \log \frac{1}{p} \to 0 \text{ per } p \to 0}
   \variante{di un esperimento con esiti $A_1, \ldots, A_N$}{H(\mathcal{E}) = \sum_{k=1}^{N} P(A_k) \log
     \frac{1}{P(A_k)}}}

\voce{entropiabinaria}{Entropia binaria}{esito binario con probabilità $p$ e $1 - p$}
  {H_2(p) = -p \log p - (1 - p) \log(1 - p)}
  {\variante{massima con la moneta onesta}{H_2\left( \frac{1}{2} \right) = 1 \text{ bit}}
   \variante{esito già noto}{H_2(0) = H_2(1) = 0}}

\voce{jensen}{Disuguaglianza di Jensen}{$g$ convessa su un intervallo che contiene i valori di $X$}
  {E[g(X)] \geq g\left( E[X] \right)}
  {\variante{$g$ concava}{E[g(X)] \leq g\left( E[X] \right)}
   \variante{uguaglianza solo se $g$ è lineare sui valori di $X$, o se $X$ è costante}{}
   \variante{senza concavità costante non c'è una relazione generale}{}}

\voce{entropiamax}{L'entropia massima}{alfabeto finito $\mathcal{X}$ con $\card{\mathcal{X}}$ simboli}
  {0 \leq H(X) \leq \log \card{\mathcal{X}}}
  {\variante{massimo con la pmf uniforme, zero solo se $X$ è costante}{}}
