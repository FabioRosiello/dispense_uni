\argomento{Media di funzioni e varianza nel continuo}{Dispensa: capitolo 13}[Media e varianza nel continuo]

\voce{valoremedioc}{Teorema fondamentale per la media}{variabile continua, se l'integrale converge}
  {E[g(X)] = \int_{-\infty}^{\infty} g(x)\, f_X(x) \dd x}
  {\variante{linearità}{E[aX + b] = a\, E[X] + b, \quad E[g_1(X) + g_2(X)] = E[g_1(X)] + E[g_2(X)]}}

\voce{varianzac}{Valore quadratico medio e varianza nel continuo}{}
  {E[X^2] = \int_{-\infty}^{\infty} x^2\, f_X(x) \dd x \qquad
   \sigma_X^2 = \int_{-\infty}^{\infty} (x - \mu_X)^2\, f_X(x) \dd x = E[X^2] - \mu_X^2}
  {\variante{valgono tutte le proprietà del discreto}{\sigma_{aX+b}^2 = a^2 \sigma_X^2, \ \text{Markov, Chebyshev}}}

\voce{quantizzazione}{Rumore di quantizzazione}{passo $\Delta$, errore uniforme su $\left(-\frac{\Delta}{2}, \frac{\Delta}{2}\right)$}
  {\sigma^2 = \frac{\Delta^2}{12}}
  {\variante{un bit in più dimezza $\Delta$}{\sigma^2 \text{ diventa } \frac{\sigma^2}{4}}}

\voce{dizionario}{Dizionario fra discreto e continuo}{le somme diventano integrali}
  {\sum_{x} (\cdots)\, p_X(x) \;\longrightarrow\; \int_{-\infty}^{\infty} (\cdots)\, f_X(x) \dd x}
  {\variante{probabilità di un insieme}{\sum_{x \in A} p_X(x) \;\longrightarrow\; \int_A f_X(x) \dd x}
   \variante{funzione $Y = g(X)$}{p_Y(y) = \sum_{g(x) = y} p_X(x) \;\longrightarrow\;
                                  f_Y(y) = \sum_{g(x_i) = y} \frac{f_X(x_i)}{\abs{\dot g(x_i)}}}
   \variante{condizionamento}{p_{X \mid A}(x) = \frac{P(\{X = x\} \cap A)}{P(A)} \;\longrightarrow\;
                              F_{X \mid A}(x) = \frac{P(\{X \leq x\} \cap A)}{P(A)}}}
