% Formulario, argomento 22: stima classica e limite di Cramér-Rao (dispensa, capitolo 22)
% ============================================================
\argomento{Stima classica e limite di Cramér-Rao}{Dispensa: capitolo 22}

\voce{verosimiglianza}{Verosimiglianza}{$\theta$ deterministico e non noto: la densità vista come funzione di
  $\theta$}
  {L(\theta; \vett{x}) = f_{\vett{X}}(\vett{x}; \theta) \qquad \Lambda(\theta; \vett{x}) = \log f_{\vett{X}}(\vett{x};
   \theta)}
  {\variante{dati indipendenti: una somma}{\Lambda(\theta; \vett{x}) = \sum_{k=1}^{n} \log f_{X_k}(x_k; \theta)}}

\voce{stimaml}{Stimatore a massima verosimiglianza (MLE)}{\emph{Maximum Likelihood Estimator}; $S$ spazio dei
  parametri}
  {\hat\theta(\vett{x}) = \arg\max_{\theta \in S} \log f_{\vett{X}}(\vett{x}; \theta)}
  {\variante{in pratica}{\frac{\partial}{\partial\theta} \log f_{\vett{X}}(\vett{x}; \theta) = 0}
   \variante{con densità a priori uniforme coincide con la stima MAP}{}
   \variante{se esiste uno stimatore efficiente, è questo; per $n$ grande è asintoticamente corretto, consistente,
     efficiente e normale}{}}

\voce{polarizzazione}{Polarizzazione (\emph{bias})}{$\hat\Theta_n$ stimatore di $\theta$ calcolato da $n$ dati}
  {E\left[ \hat\Theta_n(\vett{X}) \right] = \theta + b_n(\theta)}
  {\variante{non polarizzato (corretto)}{b_n(\theta) = 0}
   \variante{asintoticamente non polarizzato}{b_n(\theta) \to 0 \text{ per } n \to \infty}}

\voce{msebias}{Errore quadratico medio}{varianza più polarizzazione al quadrato}
  {\mathrm{MSE}\left[ \hat\Theta \right] = E\left[ \left( \hat\Theta(\vett{X}) - \theta \right)^2 \right] =
   \mathrm{VAR}\left[ \hat\Theta \right] + b_n^2(\theta)}
  {\variante{non polarizzato}{\mathrm{MSE}\left[ \hat\Theta \right] = \mathrm{VAR}\left[ \hat\Theta \right] =
     E\left[ \hat\Theta^2 \right] - \theta^2}
   \variante{consistenza: $\hat\Theta_n \to \theta$ in probabilità (debole), quasi certamente (forte), in media
     quadratica}{}}

\voce{fisher}{Informazione di Fisher}{quanto è appuntito il picco della log-verosimiglianza}
  {I_n(\theta) = E\left[ \left( \frac{\partial}{\partial\theta} \log f_{\vett{X}}(\vett{X}; \theta) \right)^2 \right] =
   -E\left[ \frac{\partial^2}{\partial\theta^2} \log f_{\vett{X}}(\vett{X}; \theta) \right]}
  {\variante{la derivata logaritmica ha media nulla}{E\left[ \frac{\partial}{\partial\theta} \log
     f_{\vett{X}}(\vett{X}; \theta) \right] = 0}
   \variante{$n$ dati i.i.d. \oltre}{I_n(\theta) = n\, I_1(\theta)}}

\voce{cramerrao}{Limite di Cramér-Rao}{nessuno stimatore può fare meglio}
  {\mathrm{VAR}\left[ \hat\Theta(\vett{X}) \right] \geq \frac{\abs{1 + b_n'(\theta)}^2}{I_n(\theta)}}
  {\variante{stimatore non polarizzato}{\mathrm{MSE}\left[ \hat\Theta \right] = \mathrm{VAR}\left[ \hat\Theta \right]
     \geq \frac{1}{I_n(\theta)}}
   \variante{stimatore qualsiasi}{\mathrm{MSE}\left[ \hat\Theta \right] \geq \frac{\abs{1 + b_n'(\theta)}^2}{I_n(\theta)}
     + b_n^2(\theta)}
   \variante{stimatore efficiente: non polarizzato e con $\mathrm{MSE} = \frac{1}{I_n(\theta)}$}{}}

\voce{mlbernoulli}{Stima della probabilità di testa}{$w$ teste in $n$ lanci di Bernoulli, $\beta$ non aleatorio}
  {\hat\beta = \frac{w}{n} \qquad E\left[ \hat\beta \right] = \beta \qquad \mathrm{VAR}\left[ \hat\beta \right] =
   \frac{\beta (1 - \beta)}{n}}
  {\variante{log-verosimiglianza}{\log p_{\vett{X}}(\vett{x}; \beta) = w \log\beta + (n - w) \log(1 - \beta)}
   \variante{informazione di Fisher}{I_n(\beta) = \frac{n}{\beta (1 - \beta)}: \text{ lo stimatore è efficiente}}}

\voce{mlgauss}{Stima della media di dati gaussiani}{$X_k$ i.i.d. $\mathcal{N}(\mu, \sigma^2)$, $\sigma^2$ noto}
  {\hat\mu = \frac{1}{n} \sum_{k=1}^{n} x_k \qquad I_n(\mu) = \frac{n}{\sigma^2} \qquad \mathrm{VAR}\left[ \hat\mu
   \right] = \frac{\sigma^2}{n} = \frac{1}{I_n(\mu)}}
  {\variante{log-verosimiglianza}{\log f_{\vett{X}}(\vett{x}; \mu) = -\frac{n}{2} \log(2\pi\sigma^2) -
     \sum_{k=1}^{n} \frac{(x_k - \mu)^2}{2\sigma^2}}
   \variante{derivate}{\frac{\partial}{\partial\mu} \log f_{\vett{X}} = \sum_{k=1}^{n} \frac{x_k - \mu}{\sigma^2},
     \quad \frac{\partial^2}{\partial\mu^2} \log f_{\vett{X}} = -\frac{n}{\sigma^2}}
   \variante{la media campione è non polarizzata ed efficiente}{}}
