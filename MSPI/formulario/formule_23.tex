\argomento{Stima lineare MMSE e gradiente}{Dispensa: capitolo 23}

\voce{lmmse}{Stimatore lineare MMSE (LMMSE)}{$\matr{M}$ covarianza dei dati, $\vett{s}$ covarianze fra dati e
  parametro}
  {\hat\Theta = \vett{a}^T \vett{X} + b: \qquad \vett{a}_{\text{LMMSE}} = \matr{M}^{-1} \vett{s} \qquad
   b_{\text{LMMSE}} = E[\Theta] - \vett{a}_{\text{LMMSE}}^T E[\vett{X}]}
  {\variante{forma equivalente}{\hat\Theta_{\text{LMMSE}} = E[\Theta] + \vett{s}^T \matr{M}^{-1} \left( \vett{X} -
     E[\vett{X}] \right)}
   \variante{il vettore $\vett{s}$}{\vett{s} = E[\Theta \vett{X}] - E[\Theta]\, E[\vett{X}] = E\left[ (\vett{X} -
     E[\vett{X}]) (\Theta - E[\Theta]) \right]}
   \variante{errore quadratico medio minimo}{E\left[ \left( \hat\Theta_{\text{LMMSE}} - \Theta \right)^2 \right] =
     \sigma_\Theta^2 - \vett{s}^T \matr{M}^{-1} \vett{s}}
   \variante{dati e parametro congiuntamente gaussiani: coincide con lo stimatore MMSE}{}}

\voce{misurarumorosa}{Una misura rumorosa}{$X = \Theta + N$, rumore indipendente a media nulla}
  {\hat\Theta = a\, X + (1 - a)\, \mu_\Theta \qquad a = \frac{\sigma_\Theta^2}{\sigma_\Theta^2 + \sigma_N^2}}
  {\variante{errore quadratico medio}{\frac{\sigma_\Theta^2\, \sigma_N^2}{\sigma_\Theta^2 + \sigma_N^2}, \quad
     \text{minore sia di } \sigma_N^2 \text{ sia di } \sigma_\Theta^2}}

\voce{gradientealg}{Algoritmo del gradiente}{$\gamma > 0$ è il passo; parte da un $\vett{a}^{(1)}$ qualsiasi}
  {\vett{a}^{(k+1)} = \vett{a}^{(k)} - \gamma \left( \matr{M} \vett{a}^{(k)} - \vett{s} \right), \qquad k = 1, 2,
   \ldots}
  {\variante{errore quadratico medio, con $b$ ottimo}{\vett{a}^T \matr{M} \vett{a} - 2\, \vett{a}^T \vett{s} +
     \sigma_\Theta^2, \quad \text{gradiente } 2\matr{M}\vett{a} - 2\vett{s}}
   \variante{errore sui coefficienti, $\vett{\epsilon}^{(k)} = \vett{a}^{(k)} - \vett{a}_{\text{LMMSE}}$}{
     \vett{\epsilon}^{(k+1)} = \left( \matr{I} - \gamma\matr{M} \right)^k \vett{\epsilon}^{(1)}}
   \variante{converge, da qualsiasi partenza, se e solo se}{0 < \gamma < \frac{2}{\lambda_{\text{MAX}}}}
   \variante{lungo l'autovettore $i$ l'errore si moltiplica a ogni passo per}{1 - \gamma \lambda_i}}
