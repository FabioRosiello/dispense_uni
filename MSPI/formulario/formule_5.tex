% ============================================================
% Formulario, argomento 5: variabili aleatorie discrete (dispensa, capitolo 5; varianze dal capitolo 7)
% ============================================================
\argomento{Variabili aleatorie discrete}{Dispensa: capitolo 5 (le varianze sono nel capitolo 7)}

\voce{pmf}{Funzione di probabilità (pmf)}{$X$ con alfabeto $\mathcal{X}$}
  {p_X(x) = P(X = x) \qquad p_X(x) \geq 0, \quad \sum_{x \in \mathcal{X}} p_X(x) = 1}
  {\variante{probabilità di un insieme di valori}{P(X \in B) = \sum_{x \in B} p_X(x)}}

\voce{mediacamp}{Media campionaria}{$n$ valori osservati}
  {\overline{X}_n = \frac{1}{n} \sum_{i=1}^{n} X(\tilde\omega_i)}
  {\variante{raggruppando i valori uguali}{\overline{X}_n = \sum_{i} x_i\, f_n(x_i)}}

\voce{media}{Media statistica (valore atteso)}{baricentro della pmf}
  {E[X] = \mu_X = \sum_{x \in \mathcal{X}} x\, p_X(x)}
  {\variante{è il limite della media campionaria}{\overline{X}_n \to E[X] \ \ \text{per } n \to \infty}}

\voce{bernoulli}{Variabile di Bernoulli}{una prova: successo (1) o insuccesso (0)}
  {p_X(1) = p, \quad p_X(0) = 1 - p}
  {\variante{media e varianza}{E[X] = p, \quad \sigma_X^2 = p\,(1 - p)}}

\voce{distbinomiale}{Variabile binomiale $\mathcal{B}(N, p)$}{successi in $N$ prove indipendenti}
  {p_X(k) = \binom{N}{k}\, p^k\, (1 - p)^{N-k}, \qquad k = 0, 1, \ldots, N}
  {\variante{media e varianza}{E[X] = N p, \quad \sigma_X^2 = N p\,(1 - p)}
   \variante{nessun successo, almeno un successo}{P(X = 0) = (1 - p)^N, \quad P(X \geq 1) = 1 - (1 - p)^N}}

\voce{uniforme}{Variabile uniforme $\mathcal{U}(\mathcal{X})$}{$M$ valori ugualmente probabili}
  {p_X(x) = \frac{1}{M} \qquad E[X] = \frac{1}{M} \sum_{x \in \mathcal{X}} x}
  {\variante{su $\{0, 1, \ldots, M - 1\}$}{E[X] = \frac{M - 1}{2}, \quad \sigma_X^2 = \frac{M^2 - 1}{12}}}

\voce{poisson}{Variabile di Poisson $\mathcal{P}(\lambda)$}{eventi rari, in media $\lambda$}
  {p_X(k) = \frac{\lambda^k}{k!}\, e^{-\lambda}, \qquad k = 0, 1, 2, \ldots}
  {\variante{media e varianza}{E[X] = \lambda, \quad \sigma_X^2 = \lambda}
   \variante{limite della binomiale}{\mathcal{B}\!\left(N, \tfrac{\lambda}{N}\right) \to \mathcal{P}(\lambda)
                                     \ \ \text{per } N \to \infty}
   \variante{$N$ grande, $p$ piccolo}{\binom{N}{k} p^k (1 - p)^{N-k} \simeq \frac{(N p)^k}{k!}\, e^{-N p}}}
