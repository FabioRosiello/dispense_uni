% Formulario, argomento 21: stima bayesiana, MMSE e MAP (dispensa, capitolo 21)
% ============================================================
\argomento{Stima bayesiana: MMSE e MAP}{Dispensa: capitolo 21}

\voce{aposteriori}{Densità a posteriori del parametro}{il parametro è una variabile $\Theta$ con densità a priori
  $f_\Theta$}
  {f_{\Theta \mid \vett{X}}(\theta \mid \vett{x}) = \frac{f_{\vett{X} \mid \Theta}(\vett{x} \mid \theta)\,
   f_\Theta(\theta)}{\int f_{\vett{X} \mid \Theta}(\vett{x} \mid \theta)\, f_\Theta(\theta) \dd\theta}}
  {\variante{dati discreti}{f_{\Theta \mid \vett{X}}(\theta \mid \vett{x}) = \frac{p_{\vett{X} \mid \Theta}(\vett{x}
     \mid \theta)\, f_\Theta(\theta)}{\int p_{\vett{X} \mid \Theta}(\vett{x} \mid \theta)\, f_\Theta(\theta)
     \dd\theta}}
   \variante{denominatore: la legge dei dati}{f_{\vett{X}}(\vett{x}) = \int f_{\vett{X} \mid \Theta}(\vett{x} \mid
     \theta)\, f_\Theta(\theta) \dd\theta}}

\voce{stimatoreottimo}{Rischio bayesiano e stimatore ottimo}{$\hat\Theta(\vett{X})$ stimatore, $\hat\theta(\vett{x})$
  stima; $C$ costo dell'errore}
  {R = E\left[ C\left( \hat\Theta(\vett{X}) - \Theta \right) \right] = E_{\vett{X}}\left[ E\left[ C\left(
   \hat\Theta(\vett{X}) - \Theta \right) \Bigm| \vett{X} \right] \right]}
  {\variante{lo stimatore ottimo, per ogni $\vett{x}$}{\hat\theta(\vett{x})_{\text{opt}} = \arg\min_{\hat\theta} \int
     C\left( \hat\theta - \theta \right) f_{\Theta \mid \vett{X}}(\theta \mid \vett{x}) \dd\theta}}

\voce{mmse}{Stimatore MMSE}{costo quadratico $C(e) = e^2$: \emph{Minimum Mean Square Error}}
  {\hat\theta_{\text{MMSE}}(\vett{x}) = \int \theta\, f_{\Theta \mid \vett{X}}(\theta \mid \vett{x}) \dd\theta =
   E\left[ \Theta \mid \vett{X} = \vett{x} \right]}
  {\variante{il baricentro della densità a posteriori}{}}

\voce{stimamap}{Stimatore MAP}{costo a tacca: gratis se l'errore è sotto $\frac{\epsilon}{2}$, 1 altrimenti}
  {\hat\theta_{\text{MAP}}(\vett{x}) = \arg\max_{\theta} f_{\Theta \mid \vett{X}}(\theta \mid \vett{x})}
  {\variante{il costo}{C(e) = \overline{\Pi}\left( \frac{e}{\epsilon} \right) = \begin{cases} 0 & \abs{e} <
     \frac{\epsilon}{2} \\ 1 & \text{altrove} \end{cases}, \quad \epsilon \to 0}
   \variante{il denominatore non dipende da $\theta$}{\hat\theta_{\text{MAP}}(\vett{x}) = \arg\max_{\theta}
     f_{\vett{X} \mid \Theta}(\vett{x} \mid \theta)\, f_\Theta(\theta)}
   \variante{il picco (la moda) della densità a posteriori; se è simmetrica con un solo picco, MAP e MMSE
     coincidono}{}}

\voce{bernoullicomposito}{La moneta misteriosa (Bernoulli composito)}{$B \sim \mathcal{U}(0, 1)$; $w$ teste in $n$
  lanci}
  {\hat\beta_{\text{MMSE}}(\vett{x}) = \frac{w + 1}{n + 2} \qquad \hat\beta_{\text{MAP}}(\vett{x}) = \frac{w}{n}}
  {\variante{verosimiglianza}{p_{\vett{X} \mid B}(\vett{x} \mid \beta) = \beta^{w} (1 - \beta)^{n - w}}
   \variante{legge dei dati}{p_{\vett{X}}(\vett{x}) = \frac{w!\, (n - w)!}{(n + 1)!}}
   \variante{densità a posteriori, $0 \leq \beta \leq 1$}{f_{B \mid \vett{X}}(\beta \mid \vett{x}) = \frac{(n +
     1)!}{w!\, (n - w)!}\, \beta^{w} (1 - \beta)^{n - w}}}

\voce{prestazionistime}{Prestazioni delle due stime della moneta}{medie fatte sui dati e sul parametro}
  {E\left[ \left( \hat\beta_{\text{MMSE}} - B \right)^2 \right] = \frac{1}{6(n + 2)} \qquad E\left[ \left(
   \hat\beta_{\text{MAP}} - B \right)^2 \right] = \frac{1}{6n}}
  {\variante{medie}{E\left[ \hat\beta_{\text{MMSE}} \right] = E\left[ \hat\beta_{\text{MAP}} \right] = \frac{1}{2}}
   \variante{momenti del peso, con la media condizionata}{E[w] = \frac{n}{2}, \quad E[w^2] = \frac{n}{6} +
     \frac{n^2}{3}}
   \variante{e ancora}{E[wB] = \frac{n}{3}, \quad E[B^2] = \frac{1}{3}}}

\voce{consistenza}{Consistenza di uno stimatore}{$\hat B_n$ calcolato da $n$ dati}
  {\text{in media quadratica:} \quad \lim_{n \to \infty} E\left[ \left( \hat B_n(\vett{X}) - B \right)^2 \right] = 0}
  {\variante{in senso debole}{\lim_{n \to \infty} P\left\{ \abs{\hat B_n(\vett{X}) - B} > \epsilon \right\} = 0
     \quad \forall\, \epsilon > 0}
   \variante{la prima implica la seconda, con Markov}{P\left\{ \abs{\hat B_n(\vett{X}) - B} > \epsilon \right\} \leq
     \frac{\mathrm{MSE}\left( \hat B_n \right)}{\epsilon^2}}}
