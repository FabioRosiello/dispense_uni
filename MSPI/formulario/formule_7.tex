% ============================================================
% Formulario, argomento 7: varianza e disuguaglianze (dispensa, capitolo 7)
% ============================================================
\argomento{Varianza e disuguaglianze}{Dispensa: capitolo 7}

\voce{rms}{Valore quadratico medio e valore efficace}{\emph{mean square}, \emph{root mean square}}
  {X_{rms}^2 = E[X^2] = \sum_{x \in \mathcal{X}} x^2\, p_X(x) \qquad X_{rms} = \sqrt{E[X^2]}}
  {}

\voce{varianza}{Varianza e deviazione standard}{la dispersione attorno alla media}
  {\sigma_X^2 = E\left[(X - \mu_X)^2\right] = \sum_{x \in \mathcal{X}} (x - \mu_X)^2\, p_X(x) \qquad
   \sigma_X = \sqrt{\sigma_X^2}}
  {}

\voce{varrms}{Varianza, valore quadratico medio e media}{il modo più veloce}
  {\sigma_X^2 = E[X^2] - E^2[X] = X_{rms}^2 - \mu_X^2}
  {\variante{al contrario}{E[X^2] = \sigma_X^2 + \mu_X^2}
   \variante{variabile a media nulla}{\sigma_X^2 = X_{rms}^2}}

\voce{momenti}{Valori quadratici medi notevoli}{e il trucco di $E[X(X-1)]$}
  {E[X^2] = E[X(X-1)] + E[X]}
  {\variante{binomiale $\mathcal{B}(N, p)$}{E[X(X-1)] = N(N-1)\, p^2, \quad E[X^2] = N p\,(1-p) + N^2 p^2}
   \variante{Poisson $\mathcal{P}(\lambda)$}{E[X(X-1)] = \lambda^2, \quad E[X^2] = \lambda + \lambda^2}
   \variante{uniforme su $\{0, \ldots, M-1\}$}{E[X^2] = \frac{(M-1)(2M-1)}{6}}}

\voce{lineare}{Media e varianza di $Y = aX + b$}{trasformazione lineare}
  {E[aX + b] = a\,E[X] + b \qquad \sigma_Y^2 = a^2 \sigma_X^2 \qquad \sigma_Y = \abs{a}\, \sigma_X}
  {\variante{la media di una costante}{E[b] = b}
   \variante{lo scarto dalla media ha media nulla}{E[X - \mu_X] = 0}
   \variante{valore quadratico medio}{E[Y^2] = a^2 X_{rms}^2 + b^2 + 2ab\, \mu_X}}

\voce{segno}{Segno della media e della varianza}{}
  {X \geq 0 \;\Rightarrow\; E[X] \geq 0 \qquad\qquad \sigma_X^2 \geq 0}
  {\variante{di conseguenza}{E[X^2] \geq E^2[X]}}

\voce{markov}{Disuguaglianza di Markov}{$Z \geq 0$, \ $\delta > 0$}
  {P(Z \geq \delta) \leq \frac{E[Z]}{\delta}}
  {}

\voce{chebyshev}{Disuguaglianza di Chebyshev}{qualsiasi pmf, $\sigma_X > 0$, $k > 0$}
  {P\left( \abs{X - \mu_X} \geq k\, \sigma_X \right) \leq \frac{1}{k^2}}
  {\variante{restare vicini alla media}{P\left( \mu_X - k\,\sigma_X \leq X \leq \mu_X + k\,\sigma_X \right) \geq 1 - \frac{1}{k^2}}
   \variante{con la distanza $\varepsilon = k\, \sigma_X$}{P\left( \abs{X - \mu_X} \geq \varepsilon \right) \leq
                                                       \frac{\sigma_X^2}{\varepsilon^2}}}

\voce{concentrazione}{Concentrazione attorno alla media}{rapporto fra media e deviazione standard}
  {\frac{\mu_X}{\sigma_X} \qquad \text{grande: variabile concentrata; piccolo: molto aleatoria}}
  {\variante{Poisson}{\frac{\mu_X}{\sigma_X} = \frac{\lambda}{\sqrt{\lambda}} = \sqrt{\lambda}}}
