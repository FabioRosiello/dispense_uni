% ============================================================
% Formulario, argomento 8: variabili aleatorie multiple (dispensa, capitolo 8)
% ============================================================
\argomento{Variabili aleatorie multiple}{Dispensa: capitolo 8}

\voce{congiunta}{Pmf congiunta}{una casella per ogni coppia $(x, y)$}
  {p_{X,Y}(x, y) = P(X = x, Y = y) = \lim_{n \to \infty} \frac{n_{X=x,\, Y=y}}{n}}
  {\variante{proprietà}{p_{X,Y}(x, y) \geq 0, \quad \sum_{x \in \mathcal{X},\, y \in \mathcal{Y}} p_{X,Y}(x, y) = 1}
   \variante{$m$ variabili}{p_{X_1, \ldots, X_m}(x_1, \ldots, x_m) = P(X_1 = x_1, \ldots, X_m = x_m)}}

\voce{marginali}{Pmf marginali}{le somme «a margine» della tabella}
  {p_X(x) = \sum_{y \in \mathcal{Y}} p_{X,Y}(x, y) \qquad p_Y(y) = \sum_{x \in \mathcal{X}} p_{X,Y}(x, y)}
  {}

\voce{indipvar}{Variabili indipendenti}{la congiunta si fattorizza}
  {p_{X,Y}(x, y) = p_X(x)\, p_Y(y) \qquad \text{per ogni } (x, y)}
  {\variante{equivalente}{p_{Y \mid X}(y \mid x) = p_Y(y), \quad p_{X \mid Y}(x \mid y) = p_X(x)}
   \variante{$m$ variabili}{p_{X_1, \ldots, X_m}(x_1, \ldots, x_m) = \prod_{i=1}^{m} p_{X_i}(x_i)}}

\voce{condpmf}{Pmf condizionate}{$p_X(x) > 0$, \ $p_Y(y) > 0$}
  {p_{Y \mid X}(y \mid x) = \frac{p_{X,Y}(x, y)}{p_X(x)} \qquad p_{X \mid Y}(x \mid y) = \frac{p_{X,Y}(x, y)}{p_Y(y)}}
  {\variante{la congiunta dalla condizionata}{p_{X,Y}(x, y) = p_{Y \mid X}(y \mid x)\, p_X(x) = p_{X \mid Y}(x \mid y)\, p_Y(y)}
   \variante{è una pmf in $y$, per ogni $x$}{\sum_{y \in \mathcal{Y}} p_{Y \mid X}(y \mid x) = 1}}

\voce{margcond}{Marginalizzazione con le condizionate}{probabilità totale per due variabili}
  {p_Y(y) = \sum_{x \in \mathcal{X}} p_{Y \mid X}(y \mid x)\, p_X(x) \qquad
   p_X(x) = \sum_{y \in \mathcal{Y}} p_{X \mid Y}(x \mid y)\, p_Y(y)}
  {}

\voce{bayespmf}{Legge di Bayes per le pmf}{}
  {p_{X \mid Y}(x \mid y) = \frac{p_{Y \mid X}(y \mid x)\, p_X(x)}{p_Y(y)}}
  {\variante{con il denominatore esplicito}
            {p_{X \mid Y}(x \mid y) = \frac{p_{Y \mid X}(y \mid x)\, p_X(x)}{\sum_{x'} p_{Y \mid X}(y \mid x')\, p_X(x')}}}

\voce{catena}{Regola della catena}{una variabile alla volta}
  {p_{X,Y,Z}(x, y, z) = p_Z(z)\; p_{Y \mid Z}(y \mid z)\; p_{X \mid Y,Z}(x \mid y, z)}
  {\variante{condizionata a più variabili}
            {p_{X \mid Y,Z}(x \mid y, z) = \frac{p_{X,Y,Z}(x, y, z)}{p_{Y,Z}(y, z)} = \frac{p_{X,Y \mid Z}(x, y \mid z)}{p_{Y \mid Z}(y \mid z)}}
   \variante{vale in qualsiasi ordine e con qualsiasi numero di variabili}{}}

\voce{indippiu}{Indipendenza di più variabili}{tutta la congiunta deve fattorizzarsi}
  {p_{X,Y,Z}(x, y, z) = p_X(x)\, p_Y(y)\, p_Z(z)}
  {\variante{non basta l'indipendenza a coppie}{}}
