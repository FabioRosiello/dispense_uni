\argomento{Variabili aleatorie continue multiple}{Dispensa: capitolo 14}[Variabili continue multiple]

\voce{congiuntac}{Densità congiunta}{probabilità per unità di area}
  {P\big( (X, Y) \in C \big) = \iint_C f_{X,Y}(x, y) \dd x \dd y}
  {\variante{definizione}{f_{X,Y}(x, y) = \lim_{\Delta x, \Delta y \to 0} \frac{P\left( \abs{X - x} \leq
                          \frac{\Delta x}{2},\ \abs{Y - y} \leq \frac{\Delta y}{2} \right)}{\Delta x\, \Delta y}}
   \variante{proprietà}{f_{X,Y}(x, y) \geq 0, \quad \int_{-\infty}^{\infty} \int_{-\infty}^{\infty} f_{X,Y}(x, y) \dd x \dd y = 1}}

\voce{marginalic}{Densità marginali}{}
  {f_X(x) = \int_{-\infty}^{\infty} f_{X,Y}(x, y) \dd y \qquad f_Y(y) = \int_{-\infty}^{\infty} f_{X,Y}(x, y) \dd x}
  {}

\voce{indipc}{Variabili continue indipendenti}{}
  {f_{X,Y}(x, y) = f_X(x)\, f_Y(y) \iff F_{X,Y}(x, y) = F_X(x)\, F_Y(y)}
  {\variante{$m$ variabili}{f_{X_1, \ldots, X_m}(x_1, \ldots, x_m) = \prod_{i=1}^{m} f_{X_i}(x_i)}}

\voce{condc}{Densità condizionata}{$f_Y(y) > 0$}
  {f_{X \mid Y}(x \mid y) = \frac{f_{X,Y}(x, y)}{f_Y(y)}}
  {\variante{è una densità in $x$}{f_{X \mid Y}(x \mid y) \geq 0, \quad \int_{-\infty}^{\infty} f_{X \mid Y}(x \mid y) \dd x = 1}}

\voce{totalec}{Probabilità totale, composta e Bayes per le densità}{}
  {f_X(x) = \int_{-\infty}^{\infty} f_{X \mid Y}(x \mid y)\, f_Y(y) \dd y}
  {\variante{composta}{f_{X,Y}(x, y) = f_Y(y)\, f_{X \mid Y}(x \mid y) = f_X(x)\, f_{Y \mid X}(y \mid x)}
   \variante{Bayes}{f_{X \mid Y}(x \mid y) = \frac{f_X(x)\, f_{Y \mid X}(y \mid x)}{f_Y(y)}}}

\voce{mediac2}{Media di una funzione di due variabili}{}
  {E[g(X, Y)] = \iint_{\R^2} g(x, y)\, f_{X,Y}(x, y) \dd x \dd y}
  {\variante{linearità}{E\left[\sum_{i=1}^{m} a_i X_i\right] = \sum_{i=1}^{m} a_i\, E[X_i]}}

\voce{mediacondc}{Teorema della media condizionata}{}
  {E[g(X, Y)] = E_Y\Big[ E_{X \mid Y}\big[ g(X, Y) \mid Y \big] \Big]}
  {\variante{con}{E[g(X, Y) \mid Y = y] = \int_{-\infty}^{\infty} g(x, y)\, f_{X \mid Y}(x \mid y) \dd x}}

\voce{covc}{Covarianza e correlazione nel continuo}{come nel discreto (\vv{covarianza}, \vv{rho})}
  {COV(X, Y) = E[XY] - \mu_X \mu_Y \qquad E[XY] = \iint_{\R^2} x\, y\, f_{X,Y}(x, y) \dd x \dd y}
  {\variante{coefficiente di correlazione}{\rho_{XY} = \frac{COV(X, Y)}{\sigma_X\, \sigma_Y}, \quad \abs{\rho_{XY}} \leq 1}
   \variante{indipendenti $\Rightarrow$ incorrelate, non il viceversa}{}}

\voce{cdfcongiunta}{CDF congiunta}{il quadrante in basso a sinistra di $(x, y)$}
  {F_{X,Y}(x, y) = P(X \leq x, Y \leq y) = \int_{-\infty}^{x} \int_{-\infty}^{y} f_{X,Y}(s, t) \dd t \dd s
   \qquad f_{X,Y} = \frac{\partial^2 F_{X,Y}}{\partial x\, \partial y}}
  {\variante{marginali}{F_X(x) = F_{X,Y}(x, +\infty), \quad F_Y(y) = F_{X,Y}(+\infty, y)}
   \variante{rettangolo $x_1 < X \leq x_2$, $y_1 < Y \leq y_2$}{P = F_{X,Y}(x_2, y_2) - F_{X,Y}(x_2, y_1) - F_{X,Y}(x_1, y_2) + F_{X,Y}(x_1, y_1)}
   \variante{limiti}{F_{X,Y}(-\infty, y) = F_{X,Y}(x, -\infty) = 0, \quad F_{X,Y}(+\infty, +\infty) = 1}}
