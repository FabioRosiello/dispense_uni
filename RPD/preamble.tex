  \ifdim\dimexpr\ht\rpdesbox+\dp\rpdesbox\relax>\textheight

    \box\rpdesbox
  \fi}

  \par\addvspace{0.8em}%

  \noindent\rule{\linewidth}{0.5pt}\par%
  \vspace{0.15em}%
  \noindent\textbf{Esempio \arabic{esempio}}%
  \ifx\relax#1\relax\else\textbf{ --- #1}\fi\par%
  \vspace{0.3em}%
}{%
  \par\vspace{0.3em}%
  \noindent\rule{\linewidth}{0.5pt}\par
  \egroup
  \rpdemetti
  \par\vspace{0.8em}%
}

  \vspace{0.15em}%

  \noindent\rule{\linewidth}{1.2pt}\par

% ============================================================
%  UN CONCETTO NON SI SPEZZA FRA DUE PAGINE
% ============================================================
% Ogni listato è chiuso in una minipage (una scatola indivisibile): se non
% entra in fondo alla pagina passa intero alla successiva. Il listato più
% lungo della dispensa ha 56 righe, quindi sta sempre in una pagina.
% La frase che introduce un listato, una figura, una tabella o una lista
% resta attaccata a ciò che introduce, e un paragrafo non si divide.
\newcommand{\rpdattacca}{\par\nopagebreak}
\AddToHook{env/lstlisting/before}{\rpdattacca\noindent\begin{minipage}{\linewidth}}
% dopo un listato si può andare a pagina nuova: due listati consecutivi
% non devono restare incollati in un blocco più alto di una pagina
\AddToHook{env/lstlisting/after}{\end{minipage}\par\penalty0 }
\AddToHook{env/figure/before}{\rpdattacca}
\AddToHook{env/table/before}{\rpdattacca}
\interlinepenalty=10000
\makeatletter
\@beginparpenalty=10000
\@itempenalty=10000
\makeatother
% Un titolo non resta da solo in fondo alla pagina
\AddToHook{cmd/section/before}{\Needspace{12\baselineskip}}
\AddToHook{cmd/subsection/before}{\Needspace{7\baselineskip}}

% Il linguaggio [POSIX]C della classe dichiara "*" come otherkeyword: listings
% consuma cosi' l'asterisco di "*/" e i commenti /* ... */ non si chiudono piu',
% colorando come commento tutto il codice successivo. Lo ridefiniamo senza "*".
\lstdefinelanguage[POSIX]{C}
                  []{}
                  { alsoletter={\#},
                    sensitive=true,
                    %% Handle directives
                    moredelim=*[s][{\color{violet!60!white}}]{\#include\ <}{.h>},
                    moredelim=*[l][{\color{blue!65!white}}]\#if,
                    moredelim=*[l][{\color{blue!65!white}}]\#endif,
                    moredelim=*[s][{\color{violet!60!white!70!black}}]{\#include\ "}{.h"},
                    moredelim=*[l][{\color{blue!90!white!70!black}}]\#define,
                    moredelim=*[s][{\color{turquoise!70!black}}]{[}{]},
                    %% Comments
                    comment=[l]//,
                    morecomment=[s]{/*}{*/},
                    % Strings
                    string=[b]{"},
                    morestring=[b]{'},
                    % Modifiers
                    keywords=[0]{const, unsigned, signed, static, volatile, typedef, struct, union, enum, atomic_t},
                    % Primitives
                    keywords=[1]{int, float, double, long, short, char, void, bool},
                    % Logical constructs
                    keywords=[2]{if, else, return, for, while, do},
                    % Operators
                    otherkeywords={&},
                    keywords=[3]{&},
                    % API structs
                    % Colors
                    directivestyle=\color{blue!85!white},
                    keywordstyle=[0]\color{cherryred},
                    keywordstyle=[1]\color{green!75!black},
                    keywordstyle=[2]\color{cherryred},
                    keywordstyle=[3]\color{naplesyellow!75!black},
                  }[keywords,comments,strings,directives]
