                               2/{System\\Design}/ph2,
                               3/{Software e\\UI Design}/ph3,

      \fi

    \foreach \i [evaluate=\i as \j using int(\i+1)] in {1,...,5}{
      \draw[-{Stealth[length=2mm]}, line width=1pt, swgray!70] (wb\i.east) -- (wb\j.west);

% Esempi ed esercizi vengono composti in una scatola: se stanno in una pagina
% la scatola va intera (eventualmente alla pagina dopo), altrimenti viene
% "svuotata" e l'esempio si spezza normalmente.
\newbox\igesbox
\newcommand{\igemetti}{%
  \ifdim\dimexpr\ht\igesbox+\dp\igesbox\relax>\textheight
    \unvbox\igesbox
  \else
    \box\igesbox
  \fi}

\renewenvironment{esempio}[1]{%
  \par\addvspace{0.8em}%
  \setbox\igesbox\vbox\bgroup
  \refstepcounter{esempio}%
  \noindent\rule{\linewidth}{0.5pt}\par%
  \vspace{0.15em}%
  \noindent\textbf{Esempio \thechapter.\theesempio}%
  \ifx\relax#1\relax\else%
    \textbf{ --- #1}%
  \fi\par%
  \vspace{0.3em}%
}{%
  \par\vspace{0.3em}%
  \noindent\rule{\linewidth}{0.5pt}\par
  \egroup
  \igemetti
  \par\vspace{0.8em}%
}

\renewenvironment{esercizio}[1]{%
  \par\addvspace{0.8em}%
  \setbox\igesbox\vbox\bgroup
  \refstepcounter{esercizio}%
  \noindent\rule{\linewidth}{1.2pt}\vspace{1pt}\par%
  \noindent\rule{\linewidth}{0.4pt}\par%
  \vspace{0.15em}%
  \noindent\textbf{Esercizio \thechapter.\theesercizio}%
  \ifx\relax#1\relax\else%
    \textbf{ --- #1}%
  \fi\par%
  \vspace{0.3em}%
}{%
  \par\vspace{0.3em}%
  \noindent\rule{\linewidth}{0.4pt}\vspace{1pt}\par%
  \noindent\rule{\linewidth}{1.2pt}\par
  \egroup
  \igemetti
  \par\vspace{0.8em}%
}
