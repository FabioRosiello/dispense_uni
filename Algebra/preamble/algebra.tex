% ── Targhette ─────────────────────────────────────────────────
\newcommand{\targhetta}[2]{%
  \tikz[baseline=(t.base)]\node[rounded corners=2pt, fill=#1!12, draw=#1!60!black, line width=0.4pt,
    font=\scriptsize\bfseries, text=#1!60!black, inner xsep=3pt, inner ysep=1.5pt] (t) {#2};}
\newcommand{\senzadim}{\targhetta{neutral}{senza dimostrazione}\par\smallskip}
\newcommand{\facoltativa}{\targhetta{neutral}{dimostrazione non richiesta dal programma}\par\smallskip}

% ── Blocchi colorati (classi di equivalenza, parti di una partizione) ──
% \blocco[larghezza]{colore}{percorso}: «macchia» colorata attorno a un gruppo di elementi.
% Il percorso unisce i centri dei nodi:
%   due elementi:  \blocco{classeA}{(n1.center) -- (n2.center)}
%   tre elementi:  \blocco{classeB}{(n1.center) -- (n2.center) -- (n3.center) -- cycle}
%   un elemento:   \blocco{classeC}{(n1.center) -- ++(0.001,0)}
\newcommand{\blocco}[3][6.4mm]{%
  \begin{scope}[on background layer]
    \draw[line width=#1+0.8pt, line cap=round, line join=round, draw=#2!55!black!45] #3;
    \draw[line width=#1, line cap=round, line join=round, draw=#2, fill=#2] #3;
  \end{scope}}

% Gli stili dei singoli nodi si cambiano con l'argomento opzionale,
% ad esempio \cubotre[k1/.style=hv evid, name prefix=p-]
% Cubo delle parti di {1, 2, 3}: nodi k0, k1, k2, k3, k12, k13, k23, k123
\NewDocumentCommand{\cubotre}{O{}}{%
  \begin{scope}[k0/.style=hv, k1/.style=hv, k2/.style=hv, k3/.style=hv,
                k12/.style=hv, k13/.style=hv, k23/.style=hv, k123/.style=hv, #1]
    \node[k0] (k0) at (0,0) {$\varnothing$};
    \node[k1] (k1) at (-1.5,1.1) {$\{1\}$}; \node[k2] (k2) at (0,1.1) {$\{2\}$}; \node[k3] (k3) at (1.5,1.1) {$\{3\}$};
    \node[k12] (k12) at (-1.5,2.2) {$\{1, 2\}$}; \node[k13] (k13) at (0,2.2) {$\{1, 3\}$}; \node[k23] (k23) at (1.5,2.2) {$\{2, 3\}$};
    \node[k123] (k123) at (0,3.3) {$\{1, 2, 3\}$};
    \draw[hlato] (k0) -- (k1); \draw[hlato] (k0) -- (k2); \draw[hlato] (k0) -- (k3);
    \draw[hlato] (k1) -- (k12); \draw[hlato] (k1) -- (k13); \draw[hlato] (k2) -- (k12);
    \draw[hlato] (k2) -- (k23); \draw[hlato] (k3) -- (k13); \draw[hlato] (k3) -- (k23);
    \draw[hlato] (k12) -- (k123); \draw[hlato] (k13) -- (k123); \draw[hlato] (k23) -- (k123);
  \end{scope}}
% Reticolo trirettangolo (d minimo; a, c, e intermedi; b massimo): nodi a, b, c, d, e
\NewDocumentCommand{\trirettangolo}{O{}}{%
  \begin{scope}[m3a/.style=hv, m3b/.style=hv, m3c/.style=hv, m3d/.style=hv, m3e/.style=hv, #1]
    \node[m3d] (d) at (0,0) {$d$};
    \node[m3c] (c) at (-1.1,1.1) {$c$}; \node[m3e] (e) at (0,1.1) {$e$}; \node[m3a] (a) at (1.1,1.1) {$a$};
    \node[m3b] (b) at (0,2.2) {$b$};
    \draw[hlato] (d) -- (c) -- (b); \draw[hlato] (d) -- (e) -- (b); \draw[hlato] (d) -- (a) -- (b);
  \end{scope}}
% Reticolo pentagonale (e < d < c < b, e < a < b): nodi a, b, c, d, e
\NewDocumentCommand{\pentagonale}{O{}}{%
  \begin{scope}[n5a/.style=hv, n5b/.style=hv, n5c/.style=hv, n5d/.style=hv, n5e/.style=hv, #1]
    \node[n5e] (e) at (0,0) {$e$};
    \node[n5d] (d) at (-1.0,0.95) {$d$}; \node[n5c] (c) at (-1.0,1.9) {$c$};
    \node[n5a] (a) at (1.0,1.4) {$a$};
    \node[n5b] (b) at (0,2.8) {$b$};
    \draw[hlato] (e) -- (d) -- (c) -- (b); \draw[hlato] (e) -- (a) -- (b);
  \end{scope}}
